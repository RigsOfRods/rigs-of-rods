list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../../cmake")

# Only discovery inputs are synthetic. Exercise the real dependency options and
# guard, without searching the host or configuring/compiling the game.
function(find_package package)
    if(package MATCHES "^(Threads|OGRE|OIS|MyGUI|fmt|RapidJSON|OpenSSL)$")
        return()
    endif()
    if(package STREQUAL "OpenAL")
        set(found_var OPENAL_FOUND)
        set(dependency_target "")
    elseif(package STREQUAL "discord_rpc")
        set(found_var discord_rpc_FOUND)
        set(dependency_target discord-rpc::discord-rpc)
    elseif(package STREQUAL "SocketW")
        set(found_var SocketW_FOUND)
        set(dependency_target SocketW::SocketW)
    elseif(package STREQUAL "Angelscript")
        set(found_var Angelscript_FOUND)
        set(dependency_target Angelscript::angelscript)
    elseif(package STREQUAL "CURL")
        set(found_var CURL_FOUND)
        set(dependency_target CURL::libcurl)
    elseif(package STREQUAL "Caelum")
        set(found_var Caelum_FOUND)
        set(dependency_target Caelum::Caelum)
    elseif(package STREQUAL "PagedGeometry")
        set(found_var PagedGeometry_FOUND)
        set(dependency_target PagedGeometry::PagedGeometry)
    else()
        message(FATAL_ERROR "Unexpected fixture package: ${package}")
    endif()
    if(TEST_MISSING_PACKAGE STREQUAL "ALL" OR TEST_MISSING_PACKAGE STREQUAL package)
        set(${found_var} OFF PARENT_SCOPE)
    else()
        set(${found_var} ON PARENT_SCOPE)
        if(dependency_target AND NOT TEST_MISSING_TARGET STREQUAL dependency_target)
            add_library(${dependency_target} INTERFACE IMPORTED)
        endif()
    endif()
endfunction()

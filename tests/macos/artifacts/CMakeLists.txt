cmake_minimum_required(VERSION 3.16)
project(macos_artifact_control NONE)
set(CMAKE_BUILD_TYPE Release)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/MacOSProfileArtifacts.cmake")
set(ROR_DEPENDENCY_DIR "${CMAKE_CURRENT_BINARY_DIR}/prefix")
file(MAKE_DIRECTORY "${ROR_DEPENDENCY_DIR}")
get_filename_component(ROR_DEPENDENCY_DIR "${ROR_DEPENDENCY_DIR}" REALPATH)
file(WRITE "${ROR_DEPENDENCY_DIR}/inside.dylib" "fixture")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/outside.dylib" "fixture")
add_library(physical SHARED IMPORTED)
set_target_properties(physical PROPERTIES IMPORTED_CONFIGURATIONS "RELEASE;DEBUG"
    IMPORTED_LOCATION_RELEASE "${ROR_DEPENDENCY_DIR}/inside.dylib"
    IMPORTED_LOCATION_DEBUG "${CMAKE_CURRENT_BINARY_DIR}/outside.dylib")
add_library(interface INTERFACE IMPORTED)
set_target_properties(interface PROPERTIES INTERFACE_LINK_LIBRARIES "${ROR_DEPENDENCY_DIR}/inside.dylib")
if(CASE STREQUAL "mapped-outside")
    set_target_properties(physical PROPERTIES MAP_IMPORTED_CONFIG_RELEASE DEBUG)
elseif(CASE STREQUAL "later-interface")
    set_property(TARGET interface APPEND PROPERTY INTERFACE_LINK_LIBRARIES "${CMAKE_CURRENT_BINARY_DIR}/outside.dylib")
elseif(CASE STREQUAL "physical-interface")
    set_target_properties(physical PROPERTIES INTERFACE_LINK_LIBRARIES "${CMAKE_CURRENT_BINARY_DIR}/outside.dylib")
elseif(CASE STREQUAL "nested-target")
    set_target_properties(physical PROPERTIES INTERFACE_LINK_LIBRARIES interface)
    set_property(TARGET interface APPEND PROPERTY INTERFACE_LINK_LIBRARIES "${CMAKE_CURRENT_BINARY_DIR}/outside.dylib")
elseif(CASE STREQUAL "unsupported-expression")
    set_target_properties(interface PROPERTIES INTERFACE_LINK_LIBRARIES "$<IF:$<BOOL:1>,${ROR_DEPENDENCY_DIR}/inside.dylib,>")
elseif(CASE STREQUAL "valid")
    set_target_properties(physical PROPERTIES MAP_IMPORTED_CONFIG_RELEASE Release
        INTERFACE_LINK_LIBRARIES "$<LINK_ONLY:interface>")
endif()
ror_macos_target(physical)
ror_macos_target(interface)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/selected.txt" CONTENT "$<TARGET_FILE:physical>")
